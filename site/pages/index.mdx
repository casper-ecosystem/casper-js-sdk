```ts [Transfer]
import {
  HttpHandler, RpcClient, NativeTransferBuilder,
  PrivateKey, KeyAlgorithm, PublicKey,
} from 'casper-js-sdk';

const rpcClient = new RpcClient(new HttpHandler('http://node:7777/rpc'));
const privateKey = PrivateKey.generate(KeyAlgorithm.ED25519);

const tx = new NativeTransferBuilder()
  .from(privateKey.publicKey)
  .target(PublicKey.fromHex('0202...b64'))
  .amount('2500000000') // 2.5 CSPR in motes
  .id(Date.now())
  .chainName('casper')
  .payment(100_000_000)
  .build();

tx.sign(privateKey);

const result = await rpcClient.putTransaction(tx);
console.log('Hash:', result.transactionHash);
```

```ts [Delegation]
import {
  HttpHandler, RpcClient, NativeDelegateBuilder,
  PrivateKey, KeyAlgorithm, PublicKey,
} from 'casper-js-sdk';

const rpcClient = new RpcClient(new HttpHandler('http://node:7777/rpc'));
const privateKey = PrivateKey.generate(KeyAlgorithm.ED25519);

const tx = new NativeDelegateBuilder()
  .from(privateKey.publicKey)
  .validator(PublicKey.fromHex('0202...b64'))
  .amount('10000000000') // 10 CSPR in motes
  .chainName('casper')
  .payment(2_500_000_000)
  .build();

tx.sign(privateKey);

const result = await rpcClient.putTransaction(tx);
console.log('Delegation hash:', result.transactionHash);
```

```ts [Contract Call]
import {
  HttpHandler, RpcClient, ContractCallBuilder,
  Args, CLValue, PrivateKey, KeyAlgorithm, PublicKey,
} from 'casper-js-sdk';

const rpcClient = new RpcClient(new HttpHandler('http://node:7777/rpc'));
const privateKey = PrivateKey.generate(KeyAlgorithm.ED25519);

const args = Args.fromMap({
  amount: CLValue.newCLUInt256('1000000000'),
  recipient: CLValue.newCLPublicKey(privateKey.publicKey),
});

const tx = new ContractCallBuilder()
  .from(privateKey.publicKey)
  .byHash('hash-abc123...')
  .entryPoint('transfer')
  .runtimeArgs(args)
  .chainName('casper')
  .payment(3_000_000_000)
  .build();

tx.sign(privateKey);

const result = await rpcClient.putTransaction(tx);
console.log('Hash:', result.transactionHash);
```
